Google Sheets Database Setup

Configure your Google Sheets integration to permanently save uploaded images and booking inquiries.

Setup Instructions (Hindi / English)

  1. Create a new Google Sheet in your Google account.
  2. In the Google Sheet menu, go to Extensions -> Apps Script.
  3. Delete any existing code in the editor and paste the code below:
// GOOGLE APPS SCRIPT BACKEND CODE
const FOLDER_NAME = "VK Event Organizer Images";

function doGet(e) {
  const action = e && e.parameter && e.parameter.action;
  if (action === "getEnquiries") {
    return handleResponse(getEnquiries());
  }
  return handleResponse(getImages());
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;
    
    if (action === "uploadImage") {
      const result = uploadImageToDrive(data.slotId, data.imageName, data.mimeType, data.base64Data);
      return handleResponse(result);
    } else if (action === "submitEnquiry") {
      const result = saveEnquiry(data);
      return handleResponse(result);
    } else {
      return handleResponse({ error: "Invalid action" });
    }
  } catch (error) {
    return handleResponse({ error: error.toString() });
  }
}

function handleResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON)
    .setHeaders({
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type"
    });
}

function getOrCreateFolder() {
  const folders = DriveApp.getFoldersByName(FOLDER_NAME);
  if (folders.hasNext()) {
    return folders.next();
  }
  const folder = DriveApp.createFolder(FOLDER_NAME);
  return folder;
}

function getOrCreateSheet(sheetName, headers) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold");
  }
  return sheet;
}

function getImages() {
  const sheet = getOrCreateSheet("Images", ["Slot ID", "Image URL", "File ID", "Timestamp"]);
  const data = sheet.getDataRange().getValues();
  const images = {};
  for (let i = 1; i < data.length; i++) {
    const slotId = data[i][0];
    const url = data[i][1];
    if (slotId && url) {
      images[slotId] = url;
    }
  }
  return { success: true, images: images };
}

function getEnquiries() {
  const sheet = getOrCreateSheet("Enquiries", [
    "Timestamp", 
    "Name", 
    "Phone", 
    "Email", 
    "Event Type", 
    "Event Date", 
    "Guests", 
    "Requirements"
  ]);
  const data = sheet.getDataRange().getValues();
  const enquiries = [];
  for (let i = 1; i < data.length; i++) {
    enquiries.push({
      timestamp: data[i][0],
      name: data[i][1],
      phone: data[i][2],
      email: data[i][3],
      eventType: data[i][4],
      eventDate: data[i][5],
      guests: data[i][6],
      requirements: data[i][7]
    });
  }
  enquiries.reverse(); // Latest enquiries first
  return { success: true, enquiries: enquiries };
}

function uploadImageToDrive(slotId, imageName, mimeType, base64Data) {
  const folder = getOrCreateFolder();
  const decoded = Utilities.base64Decode(base64Data);
  const blob = Utilities.newBlob(decoded, mimeType, imageName);
  const file = folder.createFile(blob);
  
  file.setSharing(DriveApp.Access.ANYONE, DriveApp.Permission.VIEW);
  
  const fileId = file.getId();
  const imageUrl = "https://lh3.googleusercontent.com/d/" + fileId + "=s1600";
  
  const sheet = getOrCreateSheet("Images", ["Slot ID", "Image URL", "File ID", "Timestamp"]);
  const data = sheet.getDataRange().getValues();
  let foundRow = -1;
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === slotId) {
      foundRow = i + 1;
      break;
    }
  }
  
  const now = new Date();
  if (foundRow !== -1) {
    const oldFileId = data[foundRow - 1][2];
    if (oldFileId) {
      try { DriveApp.getFileById(oldFileId).setTrashed(true); } catch (e) {}
    }
    sheet.getRange(foundRow, 2).setValue(imageUrl);
    sheet.getRange(foundRow, 3).setValue(fileId);
    sheet.getRange(foundRow, 4).setValue(now);
  } else {
    sheet.appendRow([slotId, imageUrl, fileId, now]);
  }
  
  return { success: true, imageUrl: imageUrl };
}

function saveEnquiry(data) {
  const sheet = getOrCreateSheet("Enquiries", [
    "Timestamp", 
    "Name", 
    "Phone", 
    "Email", 
    "Event Type", 
    "Event Date", 
    "Guests", 
    "Requirements"
  ]);
  
  const now = new Date();
  sheet.appendRow([
    now,
    data.name || "",
    data.phone || "",
    data.email || "",
    data.eventType || "",
    data.eventDate || "",
    data.guests || "",
    data.requirements || ""
  ]);
  
  return { success: true };
}
